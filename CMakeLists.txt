cmake_minimum_required(VERSION 3.22)
project(M7G_Golden_Voice_AI VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(M7G_HOST_JUCEAIDE "" CACHE FILEPATH "Native juceaide for cross compilation")
if(CMAKE_CROSSCOMPILING AND M7G_HOST_JUCEAIDE)
 set(JUCE_MODULES_ONLY ON CACHE BOOL "Use native helper" FORCE)
 add_subdirectory(JUCE)
 include(JUCE/extras/Build/CMake/JUCEUtils.cmake)
 add_executable(juce::juceaide IMPORTED GLOBAL)
 set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${M7G_HOST_JUCEAIDE}")
else()
 add_subdirectory(JUCE)
endif()
add_library(m7g_rnnoise STATIC
 ThirdParty/rnnoise/src/denoise.c ThirdParty/rnnoise/src/rnn.c
 ThirdParty/rnnoise/src/rnn_data.c ThirdParty/rnnoise/src/rnn_reader.c
 ThirdParty/rnnoise/src/pitch.c ThirdParty/rnnoise/src/kiss_fft.c
 ThirdParty/rnnoise/src/celt_lpc.c)
set_target_properties(m7g_rnnoise PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_include_directories(m7g_rnnoise PUBLIC ThirdParty/rnnoise/include PRIVATE ThirdParty/rnnoise/src)
if(WIN32)
 target_compile_definitions(m7g_rnnoise PRIVATE _USE_MATH_DEFINES)
else()
 target_link_libraries(m7g_rnnoise PRIVATE m)
endif()
juce_add_binary_data(AlienFonts HEADER_NAME AlienFonts.h NAMESPACE AlienFonts SOURCES Assets/Fonts/DejaVuSans.ttf Assets/Fonts/LiberationSerif-Regular.ttf)
juce_add_plugin(M7G_Golden_Voice_AI COMPANY_NAME "M7G" PLUGIN_MANUFACTURER_CODE M7G7 PLUGIN_CODE GldA FORMATS VST3 Standalone PRODUCT_NAME "M7G ALIEN VOICE" VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
target_sources(M7G_Golden_Voice_AI PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(M7G_Golden_Voice_AI PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(M7G_Golden_Voice_AI PRIVATE AlienFonts m7g_rnnoise juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
include(CTest)
if(BUILD_TESTING)
 add_executable(M7GSmoke Tests/Smoke.cpp)
 target_link_libraries(M7GSmoke PRIVATE M7G_Golden_Voice_AI m7g_rnnoise juce::juce_audio_utils juce::juce_dsp)
 add_test(NAME VocalDSP COMMAND M7GSmoke)
endif()
if(WIN32 AND BUILD_TESTING)
 add_executable(M7GVST3Load Tests/VST3Load.cpp)
 target_include_directories(M7GVST3Load PRIVATE JUCE/modules/juce_audio_processors/format_types/VST3_SDK)
 add_test(NAME VST3Module COMMAND M7GVST3Load "$<TARGET_FILE:M7G_Golden_Voice_AI_VST3>")
endif()
if(WIN32 AND BUILD_TESTING)
 add_executable(AlienUI Tests/AlienUI.cpp)
 target_link_libraries(AlienUI PRIVATE M7G_Golden_Voice_AI m7g_rnnoise juce::juce_audio_utils juce::juce_dsp)
 add_test(NAME AlienInterface COMMAND AlienUI)
endif()
if(BUILD_TESTING)
 add_executable(RackSmoke Tests/RackSmoke.cpp)
 target_link_libraries(RackSmoke PRIVATE M7G_Golden_Voice_AI m7g_rnnoise juce::juce_audio_utils juce::juce_dsp)
 add_test(NAME AlienRack COMMAND RackSmoke)
endif()
