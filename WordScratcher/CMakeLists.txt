cmake_minimum_required(VERSION 3.22)
project(WordScratcherPro300 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../JUCE7" CACHE PATH "Pinned JUCE 7.0.12")
set(M7G_HOST_JUCEAIDE "" CACHE FILEPATH "Native cross helper")
if(CMAKE_CROSSCOMPILING AND M7G_HOST_JUCEAIDE)
 set(JUCE_MODULES_ONLY ON CACHE BOOL "Native helper" FORCE)
 add_subdirectory("${JUCE_ROOT}" JUCE)
 include("${JUCE_ROOT}/extras/Build/CMake/JUCEUtils.cmake")
 add_executable(juce::juceaide IMPORTED GLOBAL)
 set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${M7G_HOST_JUCEAIDE}")
else()
 add_subdirectory("${JUCE_ROOT}" JUCE)
endif()
add_library(ScratchRubberBand STATIC ThirdParty/rubberband/single/RubberBandSingle.cpp)
target_compile_definitions(ScratchRubberBand PRIVATE USE_OWN_ALIGNED_MALLOC)
if(WIN32)
 target_compile_definitions(ScratchRubberBand PRIVATE NOMINMAX _USE_MATH_DEFINES)
endif()
target_include_directories(ScratchRubberBand PUBLIC ThirdParty/rubberband/rubberband)
set_target_properties(ScratchRubberBand PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(APPLE)
 target_link_libraries(ScratchRubberBand PUBLIC "-framework Accelerate")
endif()
juce_add_binary_data(ScratchData HEADER_NAME ScratchData.h NAMESPACE ScratchData SOURCES ../Assets/Fonts/DejaVuSans.ttf ../Assets/Fonts/RubikWetPaint-Regular.ttf)
set(SCRATCH_FORMATS VST3)
if(APPLE)
 list(APPEND SCRATCH_FORMATS AU)
endif()
juce_add_plugin(WordScratcher COMPANY_NAME "M7G" BUNDLE_ID "audio.m7g.wordscratcher.pro300" PLUGIN_MANUFACTURER_CODE M7G7 PLUGIN_CODE Ws30 FORMATS ${SCRATCH_FORMATS} PRODUCT_NAME "WORD SCRATCHER PRO - 300" VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
target_sources(WordScratcher PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(WordScratcher PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(WordScratcher PUBLIC ScratchData ScratchRubberBand juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography)
if(MSVC)
 target_compile_options(WordScratcher PUBLIC /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
 foreach(check ScratchDSP ScratchUI ScratchBenchmark)
  add_executable(${check} Tests/${check}.cpp)
  target_link_libraries(${check} PRIVATE WordScratcher)
  target_include_directories(${check} PRIVATE "${JUCE_ROOT}/modules")
  target_compile_definitions(${check} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 endforeach()
 target_compile_definitions(ScratchDSP PRIVATE SCRATCH_SPEECH_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/Tests/Assets/palabra-demo.wav")
 add_test(NAME ScratchAudio COMMAND ScratchDSP)
 add_test(NAME ScratchInterface COMMAND ScratchUI)
 if(WIN32)
  add_executable(ScratchVST3Load Tests/VST3Load.cpp)
  target_include_directories(ScratchVST3Load PRIVATE "${JUCE_ROOT}/modules/juce_audio_processors/format_types/VST3_SDK")
  add_test(NAME ScratchVST3 COMMAND ScratchVST3Load "$<TARGET_FILE:WordScratcher_VST3>")
 endif()
endif()
