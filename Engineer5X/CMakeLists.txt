cmake_minimum_required(VERSION 3.22)
project(MakkineEngineer5X VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../JUCE7" CACHE PATH "JUCE 7.0.12")
set(M7G_HOST_JUCEAIDE "" CACHE FILEPATH "Native cross-build helper")
if(APPLE)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tools/prepare-macos-juce.py" "${JUCE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/JUCE7-compatible" COMMAND_ERROR_IS_FATAL ANY)
 set(JUCE_ROOT "${CMAKE_CURRENT_BINARY_DIR}/JUCE7-compatible")
endif()
if(CMAKE_CROSSCOMPILING AND M7G_HOST_JUCEAIDE)
 set(JUCE_MODULES_ONLY ON CACHE BOOL "Native helper" FORCE)
 add_subdirectory("${JUCE_ROOT}" JUCE)
 include("${JUCE_ROOT}/extras/Build/CMake/JUCEUtils.cmake")
 add_executable(juce::juceaide IMPORTED GLOBAL)
 set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${M7G_HOST_JUCEAIDE}")
else()
 add_subdirectory("${JUCE_ROOT}" JUCE)
endif()
add_library(EngineerRubberBand STATIC ThirdParty/rubberband/single/RubberBandSingle.cpp)
target_compile_definitions(EngineerRubberBand PRIVATE USE_OWN_ALIGNED_MALLOC)
if(WIN32)
 target_compile_definitions(EngineerRubberBand PRIVATE NOMINMAX _USE_MATH_DEFINES)
endif()
set_target_properties(EngineerRubberBand PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(APPLE)
 target_link_libraries(EngineerRubberBand PUBLIC "-framework Accelerate")
endif()
juce_add_binary_data(EngineerData HEADER_NAME EngineerData.h NAMESPACE EngineerData SOURCES ../Assets/Fonts/DejaVuSans.ttf ../Assets/Fonts/RubikWetPaint-Regular.ttf)
set(ENGINEER_FORMATS VST3)
if(APPLE)
 list(APPEND ENGINEER_FORMATS AU)
endif()
juce_add_plugin(Engineer5X COMPANY_NAME "M7G" BUNDLE_ID "audio.makkine.engineer5x" PLUGIN_MANUFACTURER_CODE M7G7 PLUGIN_CODE E5x1 FORMATS ${ENGINEER_FORMATS} PRODUCT_NAME "Makkine Engineer 5X - AI Vocal Tuner Evolution" VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
target_sources(Engineer5X PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(Engineer5X PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(Engineer5X PUBLIC EngineerData EngineerRubberBand juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography)
if(MSVC)
 target_compile_options(Engineer5X PUBLIC /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
 foreach(check EngineerDSP EngineerUI)
  add_executable(${check} Tests/${check}.cpp)
  target_link_libraries(${check} PRIVATE Engineer5X)
  target_include_directories(${check} PRIVATE "${JUCE_ROOT}/modules")
  target_compile_definitions(${check} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 endforeach()
 add_test(NAME EngineerAudio COMMAND EngineerDSP)
 add_test(NAME EngineerInterface COMMAND EngineerUI)
 if(WIN32)
  add_executable(EngineerVST3Load Tests/VST3Load.cpp)
  target_include_directories(EngineerVST3Load PRIVATE "${JUCE_ROOT}/modules/juce_audio_processors/format_types/VST3_SDK")
  add_test(NAME EngineerVST3 COMMAND EngineerVST3Load "$<TARGET_FILE:Engineer5X_VST3>")
 endif()
endif()
