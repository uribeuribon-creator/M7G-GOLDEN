cmake_minimum_required(VERSION 3.22)
project(M7G_Alien_Tune VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../JUCE7" CACHE PATH "JUCE 7.0.12")
set(M7G_HOST_JUCEAIDE "" CACHE FILEPATH "Native cross helper")
set(ORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/Dependencies/onnxruntime-win-x64-1.20.1" CACHE PATH "Pinned ORT CPU SDK")
if(NOT EXISTS "${ORT_ROOT}/include/onnxruntime_cxx_api.h")
 message(FATAL_ERROR "Run Tune/Tools/Fetch-Dependencies.ps1 first")
endif()
if(CMAKE_CROSSCOMPILING AND M7G_HOST_JUCEAIDE)
 set(JUCE_MODULES_ONLY ON CACHE BOOL "Native helper" FORCE)
 add_subdirectory("${JUCE_ROOT}" JUCE)
 include("${JUCE_ROOT}/extras/Build/CMake/JUCEUtils.cmake")
 add_executable(juce::juceaide IMPORTED GLOBAL)
 set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${M7G_HOST_JUCEAIDE}")
else()
 add_subdirectory("${JUCE_ROOT}" JUCE)
endif()
juce_add_binary_data(TuneData HEADER_NAME TuneData.h NAMESPACE TuneData SOURCES ../Assets/Fonts/DejaVuSans.ttf ../Assets/Fonts/RubikWetPaint-Regular.ttf Models/AlienPitchCandidate.onnx Models/AlienVoiceProfile.onnx)
juce_add_plugin(M7G_Tune COMPANY_NAME "M7G" BUNDLE_ID "audio.m7g.alientune.god1" PLUGIN_MANUFACTURER_CODE M7G7 PLUGIN_CODE Tun1 FORMATS VST3 PRODUCT_NAME "M7G ALIEN TUNE GOD V1" VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
target_sources(M7G_Tune PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_include_directories(M7G_Tune PUBLIC "${ORT_ROOT}/include")
target_compile_definitions(M7G_Tune PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(M7G_Tune PUBLIC TuneData juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography)
add_custom_command(TARGET M7G_Tune_VST3 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:M7G_Tune_VST3>/onnxruntime.dll")
include(CTest)
if(BUILD_TESTING)
 foreach(check TuneDSP TuneUI TuneBenchmark)
  add_executable(${check} Tests/${check}.cpp)
  target_link_libraries(${check} PRIVATE M7G_Tune)
  target_include_directories(${check} PRIVATE "${JUCE_ROOT}/modules")
  target_compile_definitions(${check} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  add_custom_command(TARGET ${check} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:${check}>/onnxruntime.dll")
 endforeach()
 add_test(NAME TuneAudio COMMAND TuneDSP)
 add_test(NAME TuneInterface COMMAND TuneUI)
 if(WIN32)
  add_executable(TuneVST3Load Tests/VST3Load.cpp)
  target_include_directories(TuneVST3Load PRIVATE "${JUCE_ROOT}/modules/juce_audio_processors/format_types/VST3_SDK")
  add_test(NAME TuneVST3 COMMAND TuneVST3Load "$<TARGET_FILE:M7G_Tune_VST3>")
 endif()
endif()
