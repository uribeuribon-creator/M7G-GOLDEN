cmake_minimum_required(VERSION 3.22)
project(M7G_Alien_Transitions VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../JUCE7" CACHE PATH "Pinned JUCE 7.0.12")
set(M7G_HOST_JUCEAIDE "" CACHE FILEPATH "Native helper for cross compilation")
if(CMAKE_CROSSCOMPILING AND M7G_HOST_JUCEAIDE)
 set(JUCE_MODULES_ONLY ON CACHE BOOL "Use native helper" FORCE)
 add_subdirectory("${JUCE_ROOT}" JUCE)
 include("${JUCE_ROOT}/extras/Build/CMake/JUCEUtils.cmake")
 add_executable(juce::juceaide IMPORTED GLOBAL)
 set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${M7G_HOST_JUCEAIDE}")
else()
 add_subdirectory("${JUCE_ROOT}" JUCE)
endif()
juce_add_binary_data(TransitionsFonts HEADER_NAME TransitionsFonts.h NAMESPACE TransitionsFonts SOURCES ../Assets/Fonts/DejaVuSans.ttf ../Assets/Fonts/RubikWetPaint-Regular.ttf)
juce_add_plugin(M7G_Transitions COMPANY_NAME "M7G" BUNDLE_ID "audio.m7g.alientransitions.god1" PLUGIN_MANUFACTURER_CODE M7G7 PLUGIN_CODE Trn1 FORMATS VST3 PRODUCT_NAME "M7G ALIEN TRANSITIONS GOD V1" VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
target_sources(M7G_Transitions PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(M7G_Transitions PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(M7G_Transitions PRIVATE TransitionsFonts juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
include(CTest)
if(BUILD_TESTING)
 foreach(check TransitionsDSP TransitionsUI TransitionsBenchmark)
  add_executable(${check} Tests/${check}.cpp)
  target_link_libraries(${check} PRIVATE M7G_Transitions)
  target_include_directories(${check} PRIVATE "${JUCE_ROOT}/modules")
  target_compile_definitions(${check} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 endforeach()
 add_test(NAME TransitionAudio COMMAND TransitionsDSP)
 add_test(NAME TransitionInterface COMMAND TransitionsUI)
 if(WIN32)
  add_executable(TransitionsVST3Load Tests/VST3Load.cpp)
  target_include_directories(TransitionsVST3Load PRIVATE "${JUCE_ROOT}/modules/juce_audio_processors/format_types/VST3_SDK")
  add_test(NAME TransitionVST3 COMMAND TransitionsVST3Load "$<TARGET_FILE:M7G_Transitions_VST3>")
 endif()
endif()
