cmake_minimum_required(VERSION 3.22)
project(M7GFreshVocalPro VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../JUCE7" CACHE PATH "JUCE 7.0.12 source")
set(M7G_HOST_JUCEAIDE "" CACHE FILEPATH "Native cross compiler helper")
if(CMAKE_CROSSCOMPILING AND M7G_HOST_JUCEAIDE)
 set(JUCE_MODULES_ONLY ON CACHE BOOL "Native helper" FORCE)
 add_subdirectory("${JUCE_ROOT}" JUCE)
 include("${JUCE_ROOT}/extras/Build/CMake/JUCEUtils.cmake")
 add_executable(juce::juceaide IMPORTED GLOBAL)
 set_target_properties(juce::juceaide PROPERTIES IMPORTED_LOCATION "${M7G_HOST_JUCEAIDE}")
else()
 add_subdirectory("${JUCE_ROOT}" JUCE)
endif()
juce_add_binary_data(FreshData HEADER_NAME FreshData.h NAMESPACE FreshData SOURCES Assets/Inter-Regular.ttf Assets/Inter-Black.ttf)
if(NOT CMAKE_CROSSCOMPILING)
 set(FRESH_ICONS ICON_BIG Assets/Icon.png ICON_SMALL Assets/Icon.png)
endif()
juce_add_plugin(FreshVocal ${FRESH_ICONS} COMPANY_NAME "M7G Audio" BUNDLE_ID "com.m7g.freshvocalpro" PLUGIN_MANUFACTURER_CODE M7G7 PLUGIN_CODE FrB1 VST3_CATEGORIES Fx Dynamics Reverb FORMATS VST3 PRODUCT_NAME "M7G Fresh Vocal Pro Beast" IS_ARA_EFFECT FALSE VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
file(GLOB_RECURSE FRESH_SOURCES CONFIGURE_DEPENDS "Source/*.cpp")
target_sources(FreshVocal PRIVATE ${FRESH_SOURCES})
target_compile_definitions(FreshVocal PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(FreshVocal PUBLIC FreshData juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography juce::juce_opengl)
if(MSVC)
 target_compile_options(FreshVocal PUBLIC /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
 foreach(check FreshDSP FreshUI)
  add_executable(${check} Tests/${check}.cpp)
  target_link_libraries(${check} PRIVATE FreshVocal)
  target_include_directories(${check} PRIVATE "${JUCE_ROOT}/modules")
  target_compile_definitions(${check} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 endforeach()
 add_test(NAME FreshAudio COMMAND FreshDSP)
 add_test(NAME FreshInterface COMMAND FreshUI)
 set_tests_properties(FreshInterface PROPERTIES ENVIRONMENT "M7G_DISABLE_OPENGL=1")
 if(WIN32)
  add_executable(FreshVST3Load Tests/VST3Load.cpp)
  target_include_directories(FreshVST3Load PRIVATE "${JUCE_ROOT}/modules/juce_audio_processors/format_types/VST3_SDK")
  add_test(NAME FreshVST3 COMMAND FreshVST3Load "$<TARGET_FILE:FreshVocal_VST3>")
 endif()
endif()
